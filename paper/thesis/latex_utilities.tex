\def\@makechapterhead#1{%
  \vspace*{-0.35in}%
  {\parindent \z@ \centering \normalfont\normalsize\bfseries
    \ifnum \c@secnumdepth >\m@ne
      \@chapapp\space \thechapter
      \par\nobreak
    \fi
    #1\par\nobreak
    \vskip 40\p@
  }}
