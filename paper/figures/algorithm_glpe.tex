\begin{algorithmic}[1]
\State \textbf{Input:} on-policy transitions $\{(o_t,a_t,o_{t+1})\}_{t=1}^N$; encoder $\phi_\omega$; forward model $f_\psi$; inverse model $g_\xi$; region assignment function $\rho$; region EMAs $\mu^{\mathrm{short}}_r,\mu^{\mathrm{long}}_r$; RMS accumulators; gate states $g_r$ (optional)
\State \textbf{Output:} intrinsic rewards $\{r_t^{\mathrm{int}}\}_{t=1}^N$; updated region statistics; updated intrinsic model parameters
\For{$t=1$ to $N$}
    \State $z_t \gets \phi_\omega(o_t)$, $z_{t+1}\gets \phi_\omega(o_{t+1})$
    \State $e_t \gets \frac{1}{d}\lVert f_\psi(z_t,a_t)-z_{t+1}\rVert_2^2$
    \State Assign region $r \gets \rho(z_t)$ (inserting $z_t$ and splitting leaves when needed)
    \State Update region EMAs $\mu^{\mathrm{short}}_r,\mu^{\mathrm{long}}_r$ with $e_t$
    \State $\mathrm{LP}_t \gets \max(0,\mu^{\mathrm{long}}_r-\mu^{\mathrm{short}}_r)$
    \State $I_t \gets \lVert z_{t+1}-z_t\rVert_2$
    \State Normalize $I_t$ and $\mathrm{LP}_t$ with running RMS to get $\widetilde{I}_t,\widetilde{\mathrm{LP}}_t$
    \State $u_t \gets \alpha_{\mathrm{impact}}\widetilde{I}_t + \alpha_{\mathrm{LP}}\widetilde{\mathrm{LP}}_t$
    \If{gating enabled}
        \State Compute robust global medians $\mathrm{LP}_{\mathrm{med}}, e_{\mathrm{med}}$ across visited regions
        \State Update gate $g_r$ using persistence and hysteresis rules
        \State $r_t^{\mathrm{int}} \gets g_r\,u_t$
    \Else
        \State $r_t^{\mathrm{int}} \gets u_t$
    \EndIf
\EndFor
\State Update intrinsic model parameters $(\omega,\psi,\xi)$ using $\{(z_t,a_t,z_{t+1})\}$ and supervised losses
\end{algorithmic}
